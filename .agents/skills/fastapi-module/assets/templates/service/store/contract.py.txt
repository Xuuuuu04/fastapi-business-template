from typing import Protocol

from datetime import datetime

from shared.public import Data, meaning


# $title 可公开返回的记录信息。
class ${Camel}View(Data):
    id: int = meaning("记录编号")
    title: str = meaning("标题", rule="R-01")
    created_at: datetime = meaning("创建时间（UTC）")


# $title 步骤允许使用的数据访问操作。
class ${Camel}Store(Protocol):
    # 保存一条新记录并返回它。
    async def add(self, title: str) -> ${Camel}View: ...

    # 按编号读取记录，不存在时返回 None。
    async def find(self, record_id: int) -> ${Camel}View | None: ...
