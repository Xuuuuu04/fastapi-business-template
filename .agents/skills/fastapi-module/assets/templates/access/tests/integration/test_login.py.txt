import base64
import json

from fastapi.testclient import TestClient

from tests.testing_app import KEY

ACCOUNTS = "/api/v1/access/accounts"
LOGIN_PATH = "/api/v1/access/login"
ACCOUNT = {
    "org_id": "team",
    "username": "amy",
    "password": "correct-horse",
    "permissions": ["$name.manage"],
}
LOGIN = {"org_code": "team", "username": "amy", "password": "correct-horse", "otp_code": ""}


def create(client: TestClient) -> dict:
    reply = client.post(ACCOUNTS, json=ACCOUNT, headers=KEY)
    assert reply.status_code == 201, reply.text
    return reply.json()


def bearer(client: TestClient) -> dict[str, str]:
    reply = client.post(LOGIN_PATH, json=LOGIN)
    assert reply.status_code == 200, reply.text
    return {"Authorization": f"Bearer {reply.json()['access_token']}"}


def claims(token: str) -> dict:
    payload = token.split(".")[1]
    return json.loads(base64.urlsafe_b64decode(payload + "=" * (-len(payload) % 4)))


def test_login_returns_frontend_contract(client: TestClient) -> None:
    created = create(client)
    assert "password" not in created and created["permissions"] == ["$name.manage"]
    reply = client.post(LOGIN_PATH, json=LOGIN)
    assert reply.status_code == 200, reply.text
    body = reply.json()
    assert body["token_type"] == "Bearer" and body["org_id"] == "team"
    assert body["expires_in"] == client.app.state.settings.token_minutes * 60
    assert claims(body["access_token"])["permissions"] == ["$name.manage"]
    headers = {"Authorization": f"Bearer {body['access_token']}"}
    second = ACCOUNT | {"username": "bob", "permissions": []}
    assert client.post(ACCOUNTS, json=second, headers=headers).status_code == 201


def test_wrong_password_and_unknown_account_look_the_same(client: TestClient) -> None:
    create(client)
    wrong = client.post(LOGIN_PATH, json=LOGIN | {"password": "wrong-horse"})
    unknown = client.post(LOGIN_PATH, json=LOGIN | {"username": "nobody"})
    assert wrong.status_code == unknown.status_code == 401
    assert wrong.json()["code"] == unknown.json()["code"] == "${name}_login_failed"


def test_login_failures_are_limited(client: TestClient) -> None:
    create(client)
    for _ in range(client.app.state.settings.login_failure_limit):
        assert client.post(LOGIN_PATH, json=LOGIN | {"password": "wrong-horse"}).status_code == 401
    assert client.post(LOGIN_PATH, json=LOGIN).status_code == 429


def test_create_account_conflicts_and_limits(client: TestClient) -> None:
    create(client)
    assert client.post(ACCOUNTS, json=ACCOUNT, headers=KEY).status_code == 409
    headers = bearer(client)
    wider = ACCOUNT | {"username": "carl", "permissions": ["*"]}
    assert client.post(ACCOUNTS, json=wider, headers=headers).status_code == 403
    other = ACCOUNT | {"org_id": "other", "username": "dan", "permissions": []}
    assert client.post(ACCOUNTS, json=other, headers=headers).status_code == 403
    assert client.post(ACCOUNTS, json=ACCOUNT | {"username": "eve"}).status_code == 401
