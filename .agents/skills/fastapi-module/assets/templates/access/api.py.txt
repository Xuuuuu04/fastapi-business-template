"""
$title 的 HTTP 接口：登录、登录失败限流和签发访问令牌，新建账号交给 $name 模块。
路径固定为 /api/v1/access，请求和回包字段与前端模板 src/app/login.config.ts 的默认约定一致。
"""

import hashlib

from fastapi import APIRouter, Request
from pydantic import BaseModel, Field

from app.api.replies import responses_for
from app.api.runner import RunDep
from service.$name import public as $name
from shared.access.caller import Caller
from shared.access.tokens import issue_token
from shared.errors.framework import TooManyRequests

router = APIRouter(prefix="/api/v1/access", tags=["$title"])
FAILURE_SECONDS = 900


# 登录成功的回包：访问令牌、令牌类型、有效秒数和账号所属组织。
class LoginReply(BaseModel):
    access_token: str = Field(description="访问令牌，请求时放在 Authorization: Bearer 后面")
    token_type: str = Field(default="Bearer", description="令牌类型，固定为 Bearer")
    expires_in: int = Field(description="访问令牌还有多少秒失效")
    org_id: str = Field(description="账号所属组织代码")


# R-04：失败计数分两层，同一来源地址对同一账号、同一账号来自所有地址；成功后清零。
@router.post(
    "/login",
    summary="登录",
    operation_id="${name}_login",
    response_model=LoginReply,
    responses=responses_for($name.login),
)
async def login(body: $name.LoginInput, request: Request, run: RunDep) -> LoginReply:
    settings = request.app.state.settings
    cache = request.app.state.services.cache
    limits = failure_limits(body, request)
    for key, limit in limits.items():
        if int(await cache.get(key) or 0) >= limit:
            raise TooManyRequests("登录失败次数过多，请十五分钟后再试")
    try:
        account = await run($name.login, body)
    except $name.LoginFailed:
        for key in limits:
            await cache.incr(key, FAILURE_SECONDS)
        raise
    for key in limits:
        await cache.delete(key)
    caller = Caller(
        str(account.user_id), "user", account.username, frozenset(account.permissions), account.org_id
    )
    token, _ = issue_token(
        caller,
        settings.jwt_secret.get_secret_value(),
        settings.jwt_issuer,
        settings.jwt_audience,
        settings.token_minutes,
    )
    return LoginReply(access_token=token, expires_in=settings.token_minutes * 60, org_id=account.org_id)


# 两层失败计数的缓存键和上限；账号和地址先做摘要，缓存里不出现原文。
def failure_limits(body: $name.LoginInput, request: Request) -> dict[str, int]:
    settings = request.app.state.settings
    source = request.client.host if request.client else "unknown"
    account = f"{body.org_code}:{body.username}"
    return {
        failure_key("source", f"{account}:{source}"): settings.login_failure_limit,
        failure_key("account", account): settings.login_account_failure_limit,
    }


# 失败计数的缓存键。
def failure_key(kind: str, text: str) -> str:
    return f"login-failed:{kind}:{hashlib.sha256(text.encode()).hexdigest()}"


# 新建账号，需要 $name.manage 权限。
@router.post(
    "/accounts",
    status_code=201,
    summary="新建账号",
    operation_id="${name}_create_account",
    response_model=$name.AccountView,
    responses=responses_for($name.create_account),
)
async def create_account(body: $name.NewAccount, run: RunDep) -> $name.AccountView:
    return await run($name.create_account, body)
