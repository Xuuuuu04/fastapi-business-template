import json

from sqlalchemy import RowMapping, insert, select

from service.$name.data import AccountExists
from service.$name.store.contract import StoredAccount
from service.$name.store.tables import accounts
from shared.public import Db


class Sql${Camel}Store:
    def __init__(self, db: Db) -> None:
        self.db = db

    async def find(self, org_id: str, username: str) -> StoredAccount | None:
        query = select(accounts).where(
            accounts.c.org_id == org_id, accounts.c.username == username
        )
        row = await self.db.first(query)
        return account_of(row) if row else None

    async def add(
        self, org_id: str, username: str, password_hash: str, permissions: list[str]
    ) -> StoredAccount:
        values = {
            "org_id": org_id,
            "username": username,
            "password_hash": password_hash,
            "permissions": json.dumps(sorted(set(permissions))),
        }
        statement = insert(accounts).values(**values).returning(accounts)
        return account_of(await self.db.one(statement, conflict=AccountExists()))


def account_of(row: RowMapping) -> StoredAccount:
    return StoredAccount(
        id=row["id"],
        org_id=row["org_id"],
        username=row["username"],
        password_hash=row["password_hash"],
        permissions=json.loads(row["permissions"]),
    )
