from service.$name.store.contract import StoredAccount
from shared.public import BusinessError, Data, Kind, WorkData, meaning


class LoginInput(Data):
    org_code: str = meaning("组织代码", min_length=1, max_length=64)
    username: str = meaning("账号", min_length=1, max_length=64)
    password: str = meaning("密码", secret=True, min_length=1, max_length=256)
    otp_code: str = meaning(
        "两步验证码，当前未启用两步验证",
        empty="不使用两步验证",
        default="",
        max_length=10,
    )


class Account(Data):
    user_id: int = meaning("账号编号")
    org_id: str = meaning("所属组织代码")
    username: str = meaning("账号")
    permissions: list[str] = meaning("登录后授予调用方的权限名")


class NewAccount(Data):
    org_id: str = meaning("所属组织代码", min_length=1, max_length=64)
    username: str = meaning("账号", min_length=1, max_length=64)
    password: str = meaning("初始密码", secret=True, min_length=8, max_length=256)
    permissions: list[str] = meaning("授予的权限名", default_factory=list)


class AccountView(Data):
    user_id: int = meaning("账号编号")
    org_id: str = meaning("所属组织代码")
    username: str = meaning("账号")
    permissions: list[str] = meaning("账号拥有的权限名")


class LoginFailed(BusinessError):
    code = "${name}_login_failed"
    kind = Kind.NOT_SIGNED_IN
    message = "组织代码、账号或密码不正确"


class GrantDenied(BusinessError):
    code = "${name}_grant_denied"
    kind = Kind.FORBIDDEN
    message = "只能在自己的组织里建账号，授予的权限不能超出自己的权限"


class LoginWork(WorkData):
    org_code: str = meaning("组织代码")
    username: str = meaning("账号")
    password: str = meaning("密码", secret=True)
    otp_code: str = meaning("两步验证码，当前未启用两步验证")
    account: StoredAccount | None = meaning(
        "根据组织代码和账号查到的账号信息", empty="没有这个账号", default=None
    )
    verified: bool = meaning("密码是否正确", default=False)
    result: Account = meaning("登录成功的账号")


class CreateWork(WorkData):
    org_id: str = meaning("所属组织代码")
    username: str = meaning("账号")
    password: str = meaning("初始密码", secret=True)
    permissions: list[str] = meaning("授予的权限名")
    password_hash: str = meaning("密码哈希值", secret=True)
    result: AccountView = meaning("保存后的账号")
