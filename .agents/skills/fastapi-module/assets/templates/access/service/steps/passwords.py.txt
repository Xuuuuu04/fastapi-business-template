"""
$title 的口令计算：随机盐加 scrypt 摘要，比较用固定时间。随机盐由调用的步骤取得后传入。
"""

import hashlib
from hmac import compare_digest

# 账号不存在时拿来比较的摘要：照样算一次 scrypt，响应时间不暴露账号是否存在。
UNKNOWN_ACCOUNT = "00" * 16 + ":" + "00" * 64


# 每次约 16 MiB 内存；调用它的步骤用 heavy=True，同时计算的数量受框架准入限制。
def password_digest(password: str, salt: bytes) -> bytes:
    return hashlib.scrypt(password.encode(), salt=salt, n=2**14, r=8, p=1, dklen=64)


# 保存格式是十六进制的“盐:摘要”，口令原文不落库。
def hash_password(password: str, salt: bytes) -> str:
    return salt.hex() + ":" + password_digest(password, salt).hex()


# 用保存的盐重新计算，再固定时间比较。
def verify_password(password: str, saved: str) -> bool:
    salt, digest = saved.split(":", 1)
    actual = password_digest(password, bytes.fromhex(salt))
    return compare_digest(actual, bytes.fromhex(digest))
