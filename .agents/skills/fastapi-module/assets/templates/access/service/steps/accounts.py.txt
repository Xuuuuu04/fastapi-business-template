import secrets

from service.$name.data import AccountView, CreateWork, GrantDenied
from service.$name.steps.passwords import hash_password
from service.$name.store.contract import ${Camel}Store
from shared.public import Caller, step


@step("核对授权范围", reads=["org_id", "permissions"])
def check_grant(data: CreateWork, caller: Caller) -> None:
    same_org = caller.org_id in ("*", data.org_id)
    if not same_org or not all(caller.can(name) for name in data.permissions):
        raise GrantDenied()


@step(
    "计算口令摘要",
    reads=["password"],
    writes=["password_hash"],
    heavy=True,
)
def build_password_hash(data: CreateWork) -> None:
    data.password_hash = hash_password(data.password, secrets.token_bytes(16))


@step(
    "保存账号",
    reads=["org_id", "username", "password_hash", "permissions"],
    writes=["result"],
    saves=True,
)
async def save_account(data: CreateWork, store: ${Camel}Store) -> None:
    saved = await store.add(data.org_id, data.username, data.password_hash, data.permissions)
    data.result = AccountView(
        user_id=saved.id,
        org_id=saved.org_id,
        username=saved.username,
        permissions=saved.permissions,
    )
