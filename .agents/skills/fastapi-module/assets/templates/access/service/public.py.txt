from service.$name.data import (
    Account,
    AccountView,
    GrantDenied,
    LoginFailed,
    LoginInput,
    NewAccount,
)
from service.$name.store.contract import AccountExists
from shared.public import contract

login = contract(
    "$name.login",
    "登录",
    input=LoginInput,
    output=Account,
    access="public",
    effect="query",
    errors=(LoginFailed,),
)

create_account = contract(
    "$name.create_account",
    "新建账号",
    input=NewAccount,
    output=AccountView,
    access="$name.manage",
    effect="command",
    errors=(GrantDenied, AccountExists),
)

__all__ = [
    "Account",
    "AccountExists",
    "AccountView",
    "GrantDenied",
    "LoginFailed",
    "LoginInput",
    "NewAccount",
    "create_account",
    "login",
]
